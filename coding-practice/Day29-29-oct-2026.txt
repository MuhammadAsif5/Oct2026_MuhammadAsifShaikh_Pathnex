# Terraform

provider "aws" {
  region = "ap-southeast-2"
}

resource "aws_key_pair" "pathnex_key" {
  key_name   = "pathnex-key"
  public_key = file("~/.ssh/id_rsa.pub")
}

resource "aws_security_group" "pathnex_sg" {
  name = "pathnex-sg"

  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_instance" "pathnex_ec2" {
  ami                    = "ami-xxxxxxxx"
  instance_type          = "c5.xlarge"
  key_name               = aws_key_pair.pathnex_key.key_name
  vpc_security_group_ids = [aws_security_group.pathnex_sg.id]

  tags = {
    Name = "Pathnex-Server"
  }
}

resource "aws_ebs_volume" "pathnex_volume" {
  availability_zone = aws_instance.pathnex_ec2.availability_zone
  size              = 20
}

resource "aws_volume_attachment" "pathnex_attach" {
  device_name = "/dev/sdf"
  volume_id   = aws_ebs_volume.pathnex_volume.id
  instance_id = aws_instance.pathnex_ec2.id
}

output "public_ip" {
  value = aws_instance.pathnex_ec2.public_ip
}


# Ansible
---
- name: Install and Configure Nginx
  hosts: all
  become: yes

  tasks:

    - name: Install nginx
      yum:
        name: nginx
        state: present

    - name: Deploy index page
      copy:
        dest: /usr/share/nginx/html/index.html
        content: |
          <html>
          <h1>Welcome to Pathnex</h1>
          </html>

    - name: Start nginx
      service:
        name: nginx
        state: started
        enabled: yes

# Kubernetes

---
#PVC
apiVersion: v1
kind: PersistentVolumeClaim

metadata:
  name: pathnex-pvc

spec:
  accessModes:
    - ReadWriteOnce

  resources:
    requests:
      storage: 2Gi

---
# Deployment
apiVersion: apps/v1
kind: Deployment

metadata:
  name: pathnex-app

spec:
  replicas: 2

  selector:
    matchLabels:
      app: pathnex-app

  template:
    metadata:
      labels:
        app: pathnex-app

    spec:
      containers:
        - name: nginx
          image: nginx

          volumeMounts:
            - mountPath: /usr/share/nginx/html
              name: app-storage

      volumes:
        - name: app-storage
          persistentVolumeClaim:
            claimName: pathnex-pvc

---
# Nodeport Service
apiVersion: v1
kind: Service

metadata:
  name: pathnex-service

spec:
  type: NodePort

  selector:
    app: pathnex-app

  ports:
    - port: 80
      targetPort: 80
      nodePort: 30080   



# Shell Script
#!/bin/bash

echo "===== PATHNEX SYSTEM CHECK ====="

echo "CPU Usage:"
top -bn1 | head -5

echo
echo "Memory Usage:"
free -h

echo
echo "Disk Usage:"
df -h

echo
echo "IP Address:"
hostname -I

echo
echo "Top Process:"
ps aux --sort=-%cpu | head -2

# Jenkins Pipeline
pipeline {

    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM           = "DevOps"
        ENV            = "prod"
        PROJECT        = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {
            steps {
                sh 'terraform init'
                sh 'terraform apply -auto-approve'
            }
        }

        stage('App Deployment') {
            steps {
                sh 'ansible-playbook nginx.yml'

                sh 'kubectl apply -f pvc.yaml'
                sh 'kubectl apply -f deployment.yaml'
                sh 'kubectl apply -f service.yaml'
            }
        }

        stage('Monitoring Checks') {
            steps {

                sh '''
                echo "Institute: $INSTITUTE_NAME"
                echo "Team: $TEAM"
                echo "Environment: $ENV"
                echo "Project: $PROJECT"

                echo "CPU"
                top -bn1 | head -5

                echo "Memory"
                free -h

                echo "Disk"
                df -h

                echo "IP"
                hostname -I

                echo "Top Process"
                ps aux --sort=-%cpu | head -2
                '''
            }
        }
    }
}
