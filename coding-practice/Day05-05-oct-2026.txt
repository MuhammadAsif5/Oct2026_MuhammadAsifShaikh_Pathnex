
🔹 Jenkinsfile — With Git Integration and Testing

pipeline {
     agent any
     stages {
        stage('clone Repository') {
            steps {
                git 'https://github.com/pathnex/sample-repo.txt'
            }
        }
        
        stage('Build') {
            steps {
                echo 'Building Project from Git Repository'
            }
        }

        stage('Test') {
            steps {
                echo 'Running unit Testing'
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
     }
}

🔹 Ansible — Create User and Set Permissions

-name: Create a new User
 hosts: all
 become: yes
 tasks:
    - name: Create user
      user:
        name: pathnex
        state: present
        shell: /bin/bash

    - name: Set permissions for the user
      file:
        path: /home/pathnex
        owner: pathnex
        group: pathnex
        mode: ''0755'
