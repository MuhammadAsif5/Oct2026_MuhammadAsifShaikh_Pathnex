
# Ansible -- Create Docker Container for Multiple Applications

---
- name: Setup Docker Container for Multiple Applications
  hosts: all
  become: yes
  tasks:
    - name: Pull Web App Image
      docker_image:
        name: my-web-app 
        source: pull

    - name: Pull Database App Image
      docker_image:   
        name: my-database-app 
        source: pull

    - name: Run Web App Container
      docker_container:
         name: web-app 
         image: my-web-app 
         state: started
         published_ports:
            - "8080:80"

    - name: Run Database App Container
      docker_container:           
         name: database-app 
         image: my-database-app 
         state: started
         published_ports:
            - "3306:3306"


# Terraform -- VPC with Private and Public Subnets

resource "aws_vps" "main" {
    cidr_block = "10.0.0.0/16"
}

resource "aws_subnet" "public" {
     vpc_id     = aws_vpc.main.id
     cidr_block = "10.0.1.0/24"
     availability_zone = "us-east-1a"
     map_public_ip_on_launch = true
}

resource "aws_subnet" "private" {
     vpc_id     = aws_vpc.main.id
     cidr_block = "10.0.2.0/24"
     availability_zone = "us-east-1b"  
}

# kubernetes -- Deploying StatefulSet

apiVersion: apps/v1
kind: StatefulSet
metadata:
   name: pathnex-database
spec:
   serviceName: "pathnex-database"
   replicas: 3
   selector: 
      matchLabels:
         app: pathnex-database
   template:
      metadata:
        labels:
           app: pathnex-database
      spec:
         containers:
            - name: db 
              image: mysql:5.7
              volumeMounts:
                - mountPath: /var/lib/mysql
                  name: mysql-storage
   volumeClaimTemplates:
      - metadata:
           name: mysql-storage
        spec:
           accessModes: 
              - ReadWriteOnce
           resources:
              requests:
                storage: 1Gi

# Jenkinsfile --Build and Deploy to kubernetes

pipeline {
    agent any 
    environment {
        Image_Name = 'pathnex-web-app' 
    }
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("$(env.Image_Name)")
                }
            }
        }

        stage('Deploy to kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}

# GitLab CI/CD - Build, Push, and Deploy

stages:
    - build
    - push 
    - deploy 

build:
    stage: build
    script:
        - docker build -t pathnex-web-app .

push:
    stage: Push
    script:
        - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
        - docker push pathnex-web-app           

deploy:
    stage: deploy
    script:
        - kubectl apply -f kubernetes/deployment.yaml
