# Ansible - Create User, Add SSH Key, Set Bash Shell

---
- name: Create Pathnex Admin User
  hosts: all
  become: yes

  tasks:

    - name: Create pathnex-admin user
      user:
        name: pathnex-admin
        shell: /bin/bash
        state: present

    - name: Create .ssh directory
      file:
        path: /home/pathnex-admin/.ssh
        state: directory
        owner: pathnex-admin
        group: pathnex-admin
        mode: '0700'

    - name: Add SSH public key
      authorized_key:
        user: pathnex-admin
        state: present
        key: "{{ lookup('file', 'pathnex-admin.pub') }}"

#Jenkins

pipeline {

    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM           = "DevOps"
        ENV            = "prod"
        PROJECT        = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {
            steps {

                echo "Creating Infrastructure for ${PROJECT}"

                sh '''
                terraform init
                terraform validate
                terraform plan
                terraform apply -auto-approve
                '''
            }
        }

        stage('App Deployment') {
            steps {

                echo "Deploying Applications"

                sh '''
                ansible-playbook user-setup.yml

                kubectl apply -f deployment.yaml

                kubectl apply -f service.yaml
                '''
            }
        }

        stage('Monitoring Checks') {
            steps {

                echo "Running Monitoring Checks"

                sh '''
                
                echo "INSTITUTE : $INSTITUTE_NAME"
                echo "TEAM      : $TEAM"
                echo "ENV       : $ENV"
                echo "PROJECT   : $PROJECT"
                

                echo ""
                echo "IP ADDRESS"
                hostname -I

                echo ""
                echo "DISK USAGE"
                df -h

                echo ""
                echo "MEMORY USAGE"
                free -h
                '''
            }
        }
    }

    post {

        success {
            echo "Pipeline completed successfully."
        }

        failure {
            echo "Pipeline failed."
        }

        always {
            echo "Pathnex Training Pipeline Execution Finished."
        }
    }
}
